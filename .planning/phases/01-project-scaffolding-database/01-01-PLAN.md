    3. Create `src/lib/db.ts` — database connection utility:
       ```typescript
       import { neon } from '@neondatabase/serverless';

       if (!process.env.DATABASE_URL) {
         throw new Error('DATABASE_URL environment variable is required');
       }

       export const sql = neon(process.env.DATABASE_URL);
       ```

    4. Create `src/lib/schema.sql` — reference schema (user will run this in Neon SQL Editor):
       ```sql
       -- Expenses table
       CREATE TABLE expenses (
         id SERIAL PRIMARY KEY,
         amount NUMERIC(10, 2) NOT NULL,
         label TEXT NOT NULL,
         person TEXT NOT NULL CHECK (person IN ('Jason', 'Shelby')),
         created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
       );

       -- Settings table
       CREATE TABLE settings (
         key TEXT PRIMARY KEY,
         value TEXT NOT NULL
       );

       -- Default weekly budget
       INSERT INTO settings (key, value) VALUES ('weekly_budget', '200.00');
       ```

    5. Replace `src/app/page.tsx` with a proof-of-concept server component that queries the database:
       ```typescript
       import { sql } from '@/lib/db';

       export const dynamic = 'force-dynamic';

       export default async function Home() {
         const expenses = await sql`
           SELECT id, amount, label, person, created_at
           FROM expenses
           ORDER BY created_at DESC
         `;

         const [budget] = await sql`
           SELECT value FROM settings WHERE key = 'weekly_budget'
         `;

         return (
           <main className="min-h-dvh p-8 font-sans">
             <h1 className="text-2xl font-bold mb-4">Money Tracker</h1>
             <p className="mb-2">Weekly Budget: ${budget?.value ?? '0.00'}</p>
             <p className="mb-4">Expenses this period: {expenses.length}</p>
             {expenses.length > 0 ? (
               <ul className="space-y-1">
                 {expenses.map((e: { id: number; amount: string; label: string; person: string }) => (
                   <li key={e.id}>
                     {e.person}: {e.label} — ${e.amount}
                   </li>
                 ))}
               </ul>
             ) : (
               <p className="text-gray-500">No expenses yet.</p>
             )}
           </main>
         );
       }
       ```

    6. Create a `.env.local` placeholder file (will be overwritten by `vercel env pull`):
